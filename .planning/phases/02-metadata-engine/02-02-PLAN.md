pub type InodeId = u64;

pub struct InodeMeta {
    pub ino: u64,
    pub mode: u32,
    pub uid: u32,
    pub gid: u32,
    pub nlinks: u32,
    pub size: u64,
    pub mtime_sec: i64,
    pub mtime_nsec: u32,
    pub ctime_sec: i64,
    pub ctime_nsec: u32,
}

pub struct DirEntry {
    pub name: String,
    pub ino: InodeId,
}

pub enum MetaError {
    NotFound(InodeId),
    AlreadyExists(InodeId),
    NotADirectory(InodeId),
    IsADirectory(InodeId),
    NotEmpty(InodeId),
    InvalidName(String),
    Corrupted(String),
    Io(std::io::Error),
}

pub trait MetadataStore: Send + Sync {
    fn create_inode(&self, meta: &InodeMeta) -> Result<InodeId, MetaError>;
    fn get_inode(&self, ino: InodeId) -> Result<InodeMeta, MetaError>;
    fn update_inode(&self, meta: &InodeMeta) -> Result<(), MetaError>;
    fn delete_inode(&self, ino: InodeId) -> Result<(), MetaError>;
    fn create_directory(&self, parent_ino: InodeId, name: &str, meta: &InodeMeta) -> Result<InodeId, MetaError>;
    fn list_directory(&self, ino: InodeId) -> Result<Vec<DirEntry>, MetaError>;
    fn lookup(&self, parent_ino: InodeId, name: &str) -> Result<InodeId, MetaError>;
    fn link(&self, parent_ino: InodeId, name: &str, ino: InodeId) -> Result<(), MetaError>;
    fn unlink(&self, parent_ino: InodeId, name: &str) -> Result<(), MetaError>;
    fn set_manifest(&self, ino: InodeId, blocks: &[Digest224]) -> Result<(), MetaError>;
    fn get_manifest(&self, ino: InodeId) -> Result<Vec<Digest224>, MetaError>;
    fn set_xattr(&self, ino: InodeId, name: &str, value: &[u8]) -> Result<(), MetaError>;
    fn get_xattr(&self, ino: InodeId, name: &str) -> Result<Vec<u8>, MetaError>;
    fn list_xattrs(&self, ino: InodeId) -> Result<Vec<String>, MetaError>;
    fn remove_xattr(&self, ino: InodeId, name: &str) -> Result<(), MetaError>;
    fn root_ino(&self) -> InodeId;
    fn commit(&self) -> Result<Digest224, MetaError>;
}

  2. Create `crates/metadata/src/directory.rs`:
     - Directory entries are stored as a list of (name, ino) pairs serialized into a CAS tree.
     - Encoding: each entry = length-prefixed name bytes + 8-byte LE ino. Full directory = concatenation of all entries stored via State::push_all.
     - `pub fn create_dir_entries(dict: &mut impl StorageAdd, self_ino: u64, parent_ino: u64) -> Digest224` -- creates directory with . and .. entries
     - `pub fn list_dir_entries(dict: &impl StorageGet, dir_digest: &Digest224) -> Result<Vec<DirEntry>, MetaError>` -- deserializes all entries
     - `pub fn add_dir_entry(dict: &mut impl StorageAdd, dir_digest: &Digest224, name: &str, ino: u64) -> Result<Digest224, MetaError>` -- loads entries, adds new one, re-serializes. Returns new dir digest. Returns MetaError::AlreadyExists if name exists.
     - `pub fn remove_dir_entry(dict: &mut impl StorageAdd + StorageGet, dir_digest: &Digest224, name: &str) -> Result<Digest224, MetaError>` -- loads entries, removes named one, re-serializes. Returns MetaError::NotFound if name missing. Rejects removing "." or "..".
     - `pub fn lookup_dir_entry(dict: &impl StorageGet, dir_digest: &Digest224, name: &str) -> Result<u64, MetaError>` -- finds entry by name, returns ino
     - Entry serialization format: `[name_len: u32 LE][name_bytes: name_len bytes][ino: u64 LE]` concatenated for all entries.
     - NOTE: This is a simpler approach than per-entry CAS subtrees (from CONTEXT.md discretion). For Phase 2 correctness, linear serialization is sufficient. Per-entry subtrees can be optimized in later phases if directories grow large. The important thing is the function signatures remain stable.
     - Unit tests for create, add, remove, lookup, . and .. presence

     - `pub fn load_manifest(dict: &impl StorageGet, key: &Digest224) -> Result<Vec<Digest224>, MetaError>` -- get bytes, deserialize 28-byte chunks
     - Return MetaError::Corrupted if bytes length is not a multiple of 28

       - Create empty Dictionary
       - Create InodeMap (next_ino starts at 2)
       - Create root directory inode (ino=1, mode=0o40755, uid=0, gid=0, nlinks=2)

       - Create root dir entries (. -> 1, .. -> 1) in dict
       - Insert ino 1 into inode_map and inode_data/dir_data maps
     - Implement `MetadataStore` trait:
       - `create_inode`: allocate ino from inode_map, set meta.ino to allocated value, intern into dict, store in inode_data map, insert into inode_map. Return allocated ino.
