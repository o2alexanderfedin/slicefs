```rust
use std::sync::{Arc, Mutex};
use blockset::file_storage_get;
use metadata::store::DictMetadataStore;
use metadata::store_io::StoreIo;
use slicefs_cli::filesystem::SliceFsFilesystem;

fn fresh_fs() -> (SliceFsFilesystem, tempfile::TempDir) {
    let dir = tempfile::tempdir().unwrap();
    let io = Arc::new(Mutex::new(StoreIo::new(dir.path())));
    let meta = DictMetadataStore::new(io.clone());
    let fs = SliceFsFilesystem::new(meta, io, None);
    (fs, dir)
}
```

**Test 1 — test_write_raw_no_compression (DECOMP-01, DECOMP-02):**

- Release both
- Get manifest digests for both files
- Assert: both digests are identical (same raw content = same Digest224 = dedup)

Note: The test helper uses `simulate_*` methods if available on SliceFsFilesystem. If not, use the FUSE callback methods directly (create, write, release) with appropriate Request mocking — follow the pattern established in write_path_tests.rs.
